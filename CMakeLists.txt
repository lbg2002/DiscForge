cmake_minimum_required(VERSION 3.16)
project(DiscForge VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Concurrent Test DBus)
add_library(discforge_core src/core.cpp src/iso_reader.cpp)
target_link_libraries(discforge_core PUBLIC Qt6::Core Qt6::DBus)
target_include_directories(discforge_core PUBLIC src)
target_compile_options(discforge_core PRIVATE -Wall -Wextra -Wpedantic)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DISCFORGE_VERSION LIMIT_COUNT 1)
if(NOT DISCFORGE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[A-Za-z0-9.]+)?$")
 message(FATAL_ERROR "Invalid VERSION")
endif()
configure_file(src/version.h.in generated/version.h @ONLY)
add_executable(discforge src/main.cpp src/window.cpp)
target_include_directories(discforge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(discforge PROPERTIES OUTPUT_NAME discforge-native)
target_link_libraries(discforge PRIVATE discforge_core Qt6::Widgets Qt6::Concurrent)
target_compile_options(discforge PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS discforge RUNTIME DESTINATION bin)
install(FILES LICENSE DESTINATION share/doc/discforge RENAME copyright)
install(FILES packaging/io.github.discforge.native.desktop DESTINATION share/applications)
install(FILES packaging/discforge.svg DESTINATION share/icons/hicolor/scalable/apps)
include(CTest)
if(BUILD_TESTING)
 add_executable(lock_probe tests/lock_probe.cpp)
 add_executable(core_tests tests/core_tests.cpp)
 add_dependencies(core_tests lock_probe)
 target_link_libraries(core_tests PRIVATE discforge_core Qt6::Test)
 add_test(NAME core COMMAND core_tests)
 add_test(NAME interface COMMAND discforge --self-test)
 set_tests_properties(interface PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
